package validation

import (
    "context"
    "testing"
    mariadb "github.com/aantich/go-mariadb"
)

func TestReviewDescribeBooleanConsistency(t *testing.T) {
    cfg := driverConfig(t, "text")
    if err := cfg.Apply(mariadb.WithExtendedMetadata(true), mariadb.TinyInt1IsBool(true)); err != nil { t.Fatal(err) }
    db := openConfig(t, cfg)
    conn, err := db.Conn(context.Background())
    if err != nil { t.Fatal(err) }
    defer conn.Close()
    mustExec(t, conn, "CREATE TEMPORARY TABLE go_review_stage2_bool(v TINYINT(1) NOT NULL)")
    description, err := mariadb.Describe(t.Context(), conn, "SELECT v FROM go_review_stage2_bool")
    if err != nil { t.Fatal(err) }
    rows, err := conn.QueryContext(t.Context(), "SELECT v FROM go_review_stage2_bool")
    if err != nil { t.Fatal(err) }
    defer rows.Close()
    columns, err := rows.ColumnTypes()
    if err != nil { t.Fatal(err) }
    t.Logf("Describe=%s, ColumnTypes=%s, ScanType=%s", description[0].DatabaseType, columns[0].DatabaseTypeName(), columns[0].ScanType())
    if description[0].DatabaseType != columns[0].DatabaseTypeName() {
        t.Errorf("public metadata helpers disagree for the same configured connection and query")
    }
}
