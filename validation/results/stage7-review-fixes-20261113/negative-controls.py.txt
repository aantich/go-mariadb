"""Restore the reviewed defects in memory; permanent regressions must reject them."""
import ast
import contextlib
import io
import json
from pathlib import Path
import subprocess
import sys
import unittest
from unittest import mock

sys.path.insert(0,'validation')
import compatibility as matrix
import test_compatibility

out=Path('validation/results/stage7-review-fixes-20261113')
previous=ast.parse(subprocess.check_output(['git','show','2a5fe58:validation/compatibility.py'],text=True))
def old_function(name,owner=None):
    nodes=previous.body
    if owner:nodes=next(node for node in nodes if isinstance(node,ast.ClassDef) and node.name==owner).body
    node=next(node for node in nodes if isinstance(node,ast.FunctionDef) and node.name==name)
    namespace=matrix.__dict__
    sentinel=object();original=namespace.get(name,sentinel)
    exec(compile(ast.Module(body=[node],type_ignores=[]),'<reviewed original '+name+'>','exec'),namespace)
    function=namespace[name]
    if original is sentinel:del namespace[name]
    else:namespace[name]=original
    return function

controls=[
    ('restore-growing-historical-floor',test_compatibility.CompatibilityEvidenceTests,
     'test_inventory_growth_preserves_older_passes_and_commands',[(matrix,'validate',old_function('validate'))]),
    ('restore-stage7-only-discovery',test_compatibility.CompatibilityEvidenceTests,
     'test_prior_profile_and_newer_passing_leaf_cannot_be_omitted',[(matrix,'prior_profile',old_function('prior_profile'))]),
    ('restore-lost-setup-diagnostic',test_compatibility.CompatibilityOwnershipTests,
     'test_setup_failure_retains_redacted_docker_diagnostics',
     [(matrix,'run',old_function('run')),(matrix.OwnedServer,'docker',old_function('docker','OwnedServer'))]),
]
results={}
for name,case,test,patches in controls:
    stream=io.StringIO()
    with contextlib.ExitStack() as stack,contextlib.redirect_stdout(stream):
        for owner,attribute,value in patches:stack.enter_context(mock.patch.object(owner,attribute,value))
        result=unittest.TextTestRunner(stream=stream,verbosity=2).run(unittest.TestSuite([case(test)]))
    (out/(name+'.log')).write_text(stream.getvalue())
    assert result.testsRun==1 and not result.wasSuccessful(),name
    results[name]={'tests_run':result.testsRun,'failures':len(result.failures),'errors':len(result.errors),
                   'restored_defect_rejected':True}
(out/'negative-controls.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(results,indent=2))
