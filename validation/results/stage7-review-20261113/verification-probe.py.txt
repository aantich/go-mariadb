"""Check whether the standard verifier reads the new compatibility pointer."""
import contextlib
import io
import json
from pathlib import Path
import sys
from unittest import mock

sys.path.insert(0, 'validation')
import record
import verify

target = record.HERE / 'results/compatibility-current.json'
read_text = Path.read_text
exists = Path.exists
results = {}
for scenario in ('missing_compatibility_report', 'stale_incomplete_compatibility_report'):
    reads = []
    def fake_read(path, *args, **kwargs):
        if path == target:
            reads.append(str(path))
            if scenario == 'missing_compatibility_report':
                raise FileNotFoundError(str(path))
            return json.dumps({'inputs': {}, 'profiles': {}, 'profile_count': 0})
        return read_text(path, *args, **kwargs)
    def fake_exists(path):
        if path == target and scenario == 'missing_compatibility_report':
            return False
        return exists(path)
    output = io.StringIO()
    with mock.patch.object(Path, 'read_text', fake_read), \
         mock.patch.object(Path, 'exists', fake_exists), contextlib.redirect_stdout(output):
        verify.main()
    results[scenario] = {'verifier_succeeded': True,
                         'compatibility_reads': len(reads),
                         'verifier_output': output.getvalue().strip()}
    assert not reads
print(json.dumps(results, indent=2))
