import ast,hashlib,json,re,subprocess,sys
from pathlib import Path
sys.path.insert(0,'validation')
import compatibility,correctness,record
ROOT=Path.cwd();OUT=Path('validation/results/stage7-20261112')
def digest(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
source=json.loads((OUT/'source.json').read_text())
assert record.inputs()==source['inputs']
cohort=json.loads(Path('validation/results/compatibility-current.json').read_text())
assert cohort['inputs']==source['inputs'] and cohort['profile_count']==12

    assert digest(directory/'capture.json')==p['capture_sha256']

profiles={}
for name in ('mac','linux'):
    directory=Path('validation/results')/('stage7-'+name+'-final-3')
    capture=json.loads((directory/'capture.json').read_text())
    summary=correctness.summarize(capture,directory)
    prior=json.loads((Path('validation/results')/('stage6-review-fixes-'+name+'-final/capture.json')).read_text())
    before={k for k,v in correctness.leaves(prior['stages']['correctness']['cases']).items() if v.get('status')=='pass'}
    after={k for k,v in correctness.leaves(capture['stages']['correctness']['cases']).items() if v.get('status')=='pass'}
    assert not before-after and len(after-before)==49
    assert 'Ran 99 tests' in (directory/'recorder.log').read_text()

        'leaf_counts':summary['leaf_counts'],'prior_stage6_passing_leaves':len(before),'new_passing_leaves':49,
        'prior_passes_lost':[],'checks':summary['verified_checks'],'harness_tests':99,
        'go':capture['go'],'host':capture['host'],'server':capture['server'],'cdc_fixture':capture['cdc_fixture']}
first=json.loads((OUT/'source-first.json').read_text())
differences={kind:sorted(path for path in set(first['inputs'][kind])|set(source['inputs'][kind])
            if first['inputs'][kind].get(path)!=source['inputs'][kind].get(path)) for kind in source['inputs']}
assert differences=={'driver':[],'workload':['validation/compatibility.py','validation/test_compatibility.py']}

files=[p for p in files if not p.startswith('validation/results/')]

subprocess.run(['/private/tmp/gomdb-review-tools/actionlint'],check=True)

def passwords(value):
    if isinstance(value,dict):
        return [v for k,v in value.items() if 'pass' in k.lower() and isinstance(v,str) and v]+[p for v in value.values() for p in passwords(v)]
    if isinstance(value,list):return [p for v in value for p in passwords(v)]
    return []
secrets=passwords(json.loads(Path('validation/.local/config.json').read_text()));assert secrets
paths=[p for directory in Path('validation/results').glob('stage7*') for p in (directory.rglob('*') if directory.is_dir() else [directory]) if p.is_file() and p.name!='stage7-confirmation.json']
