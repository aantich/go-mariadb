"""Run every declared compiler/server cell; never select a favorable retry."""
import json,os,subprocess,sys
from pathlib import Path
root=Path.cwd();out=root/'validation/results/stage7-matrix-final';out.mkdir(exist_ok=False)
toolchains={
    '1.27':'/opt/homebrew/bin/go',
    '1.26':'/private/tmp/go-mariadb-stage7-toolchains-20261112/1.26/go/bin/go',
    '1.25':'/private/tmp/go-mariadb-stage7-toolchains-20261112/1.25/go/bin/go'}
env={**os.environ,'PYTHONDONTWRITEBYTECODE':'1','GOCACHE':'/private/tmp/go-mariadb-validation-20261111/go-cache','GOPROXY':'off'}
results=[]
for compiler,binary in toolchains.items():
    for family in ('10.11','11.4','11.8','12.3'):
        directory=out/('go'+compiler+'-mariadb'+family)
        command=[sys.executable,'validation/compatibility.py','--family',family,'--go',binary,'--output',str(directory)]
        with (out/(directory.name+'-launch.log')).open('w') as log:
            result=subprocess.run(command,env=env,stdout=log,stderr=subprocess.STDOUT,timeout=1200)
        item={'compiler':compiler,'family':family,'command':command,'exit_code':result.returncode,'directory':str(directory.relative_to(root))}
        results.append(item);print(json.dumps(item),flush=True)
        (out/'matrix.json').write_text(json.dumps(results,indent=2)+'\n')
        if result.returncode:print('Cell failed; retaining it and continuing the other cells.',flush=True)
raise SystemExit(1 if any(item['exit_code'] for item in results) else 0)
